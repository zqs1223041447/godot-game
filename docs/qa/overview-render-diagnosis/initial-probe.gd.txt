func cpu_stats()->Dictionary:
	var result:Dictionary={}
	for line:String in FileAccess.get_file_as_string("/sys/fs/cgroup/cpu.stat").split("\n",false):
		var parts:=line.split(" ",false)
		if parts.size()==2:result[parts[0]]=int(parts[1])
	return result
