var checks: int = 0

var rejection_cases: int = 0
var failed_labels: Array[String] = []

var special_templates: Dictionary = {}
var report: Dictionary = {}


func check(ok: bool, label: String) -> void:
	checks += 1
	if not ok:
		failures += 1
		failed_labels.append(label)
		push_error(label)


func _initialize() -> void:
	call_deferred("_run")


func _run() -> void:
	if not OS.get_environment("XDG_DATA_HOME").begins_with("/tmp/godot-m1-v116-compat-camp-"):
		printerr("COMPAT_CAMP_BLOCKED: isolated XDG directory required"); quit(78); return
	var selections: Array = [[]]
	for id: String in Modifiers.get_ids(): selections.append([id])
	selections.append(["enemy_max_health_120", "enemy_shield_from_health_20"])
	selections.append(["enemy_damage_115", "enemy_attack_speed_110"])
	selections.append(["enemy_armour_80", "enemy_move_speed_110"])
	# The original four-map loop body and all legacy assertions stay unchanged.
	# Native-only catalogue entries have their own exercised contract below.
	for map_id: String in LEGACY_MAP_IDS:
		for normal_ids: Array in selections:
			for special_ids: Array in [[], ["elemental_aegis"], ["frost_patrol"], ["storm_patrol"]]:
				var compiled: Dictionary = Maps.compile(map_id, normal_ids, special_ids)
				if compiled.ok: _positive(compiled.profile)
		for tier: int in range(1, 4):
			_positive(Maps.compile_normal(map_id, tier, [], []).profile)
			var normal: Dictionary = Maps.compile_normal(map_id, tier, ["enemy_max_health_120", "enemy_shield_from_health_20"], ["elemental_aegis"])
			if normal.ok: _positive(normal.profile)
		_failures(Maps.compile(map_id, [], []).profile)
	_boundary_checks()
	report["legacy"] = {"map_ids": LEGACY_MAP_IDS, "checks": checks, "failures": failures,
		"staged_groups": positive_groups, "rejected_groups": rejection_cases}
	await _native_checks(selections)
	report.merge({"checks": checks, "failures": failures, "failed_labels": failed_labels,
		"staged_groups": positive_groups, "rejected_groups": rejection_cases,
		"native_groups": native_groups, "native_profiles": native_profiles,
		"native_rejections": native_rejections, "special_templates": special_templates,
		"method": "Unchanged four-map camp cases plus real detached native assembly, readiness, routes, PreparedMapEntry ownership and factory groups; no Main, save, battle ticks or rendering"})
	var output := FileAccess.open(QA, FileAccess.WRITE)
	if output != null:
		output.store_string(JSON.stringify(report, "\t") + "\n"); output.close()
	print("COMPAT_CAMP_RESULT " + JSON.stringify(report))
	print("Map camp admission: %d checks, %d failures; %d staged groups, %d rejected groups" % [checks, failures, positive_groups, rejection_cases])
	quit(1 if failures else 0)


func _runtime() -> RefCounted:
	var runtime = Runtime.new()
	var parent: Dictionary = runtime.create_root("splitter", 5, Vector2(150, 250))
	parent.health = 0.0
	runtime.process_death(parent)
	runtime.create_root("brute", 5, Vector2(350, 250))
	return runtime


func _geometry(map_id: String) -> RefCounted:
	var geometry = Geometry.new()
	check(geometry.configure(map_id, View.WORLD_ARENA), "Fixture configures the actual map geometry")
	if geometry.has_walls():
		geometry.direction(View.WORLD_ARENA.position + Vector2(100, 350), View.WORLD_ARENA.position + Vector2(1740, 350), 14.0)
	return geometry


func _entries(profile: Dictionary) -> Array[Dictionary]:
	var entries: Array[Dictionary] = []
	var templates: Array[String] = ["crawler", "skitter", "brute", "splitter", "brood_host", "brute", "skitter", "ember_guard"]
	var count: int = 8 if profile.id == "old_garden" else 12
	for index: int in range(count):
		var template_id: String = templates[index % templates.size()]
		var source: Dictionary = Monsters.TEMPLATES[template_id]
		var roll: Dictionary = {"template": template_id, "rarity": source.rarity, "mechanisms": source.mechanisms.duplicate(true)}
		if template_id == "brute": roll.rarity = "magic"; roll.mechanisms = ["gale_stride"]
		if template_id == "skitter": roll.rarity = "rare"; roll.mechanisms = ["ember_power", "aegis_capacity"]
		var special: String = Maps.special_template(profile, roll)
		if not special.is_empty(): template_id = special
		entries.append({"template_id": template_id, "rarity": roll.rarity, "mechanisms": roll.mechanisms,
			"position": View.WORLD_ARENA.position + Vector2(120 + (index % 4) * 80, 140 + (index / 4) * 80), "admission_index": index + 1})
	return entries


func _boss_entry(profile: Dictionary) -> Array[Dictionary]:
	return [{"template_id": profile.boss_id, "rarity": "", "mechanisms": [], "position": View.WORLD_ARENA.position + Vector2(140, 340)}]


func _preserving(runtime: RefCounted, profile: Variant, entries: Variant, geometry: RefCounted,
		player: Variant, available: Variant, context: String = "ordinary", prepared: Variant = null) -> Dictionary:
	var before: PackedByteArray = var_to_bytes(Encounter._snapshot(runtime))
	var templates_before: PackedByteArray = var_to_bytes(runtime.templates)
	var errors_before: PackedByteArray = var_to_bytes(runtime.validation_errors)
	var entries_before: PackedByteArray = var_to_bytes(entries)
	var profile_before: PackedByteArray = var_to_bytes(profile)
	var geometry_before: PackedByteArray = var_to_bytes([geometry.snapshot(), geometry._routes])
	var player_before: PackedByteArray = var_to_bytes(player)
	seed(430003)
	var expected_random: Array[int] = [randi(), randi(), randi()]
	seed(430003)
	var result: Dictionary = Camps.plan(runtime, profile, entries, geometry, player, available, context, prepared)
	check([randi(), randi(), randi()] == expected_random, "Plan leaves the global RNG stream untouched")
	check(var_to_bytes(Encounter._snapshot(runtime)) == before, "Plan preserves the exact original runtime checkpoint")
	check(var_to_bytes(runtime.templates) == templates_before and var_to_bytes(runtime.validation_errors) == errors_before, "Plan preserves original templates and validation state")
	check(var_to_bytes(entries) == entries_before and var_to_bytes(profile) == profile_before, "Plan preserves entries and the compiled profile byte for byte")
	check(var_to_bytes([geometry.snapshot(), geometry._routes]) == geometry_before and var_to_bytes(player) == player_before, "Plan preserves geometry, routes, and player position")
	return result


func _positive(profile: Dictionary) -> void:
	var geometry: RefCounted = _geometry(profile.id)
	for context: String in ["ordinary", "map_boss"]:
		var runtime: RefCounted = _runtime()
		var entries: Array[Dictionary] = _entries(profile) if context == "ordinary" else _boss_entry(profile)
		var before: Dictionary = Encounter._snapshot(runtime)
		var result: Dictionary = _preserving(runtime, profile, entries, geometry, View.WORLD_ARENA.get_center(), entries.size(), context)
		check(result.ok and result.error.is_empty() and result.enemies.size() == entries.size(), "Whole group stages with exact capacity: " + profile.summary + " / " + context)
		if not result.ok: continue
		positive_groups += 1
		var expected = Runtime.new(runtime.templates)
		Encounter._restore(expected, before.duplicate(true))
		var seen: Dictionary = {}
		for index: int in range(entries.size()):
			var entry: Dictionary = entries[index]
			var enemy: Dictionary = result.enemies[index]
			var reference: Dictionary = Admission.create_root(expected, profile, entry.template_id, profile.wave, entry.position, context, entry.rarity, entry.mechanisms, true)
			check(reference.ok and var_to_bytes(reference.enemy) == var_to_bytes(enemy), "Every named actor field equals the existing map admission source")
			check(enemy.id == before.next_id + index + 1 and not seen.has(enemy.id), "Staged identities are sequential and unique")
			seen[enemy.id] = true
			check(enemy.pos == entry.position and enemy.spawn == 0.6 and enemy.wave == profile.wave, "Exact position, wave, and catalog spawn delay remain unchanged")
			var expected_radius: float = float(Monsters.SPECIES[enemy.kind].radius) * (1.25 if context == "map_boss" else 1.0)
			check(enemy.radius == expected_radius and enemy.radius <= (27.5 if context == "map_boss" else 22.0), "Geometry uses actual species/boss radius")
			check(enemy.root_id == enemy.id and enemy.generation == 0 and enemy.reward_eligible, "Every staged actor is a reward-eligible root")
			if not profile.normal_ids.is_empty(): check(enemy.has("encounter_source"), "Normal modifiers retain their actual source metadata")
			if profile.special_ids.has("elemental_aegis"): check(enemy.has("map_defense_source"), "Aegis retains its actual source metadata")
			if context == "map_boss": check(enemy.map_boss_attack_id == profile.boss_attack_id and enemy.template_id == profile.boss_id, "Boss template and named map attack use the actual profile")
		check(var_to_bytes(result.runtime_checkpoint) == var_to_bytes(Encounter._snapshot(expected)), "Returned checkpoint exactly matches existing factory authority")
		check(result.runtime_checkpoint.queue == before.queue and result.runtime_checkpoint.trace == before.trace, "Pending descendants and existing trace are retained")
		for id: int in before.roots: check(result.runtime_checkpoint.roots[id] == before.roots[id], "Existing roots and processed lineage remain intact")
		Encounter._restore(runtime, result.runtime_checkpoint.duplicate(true))
		var repeated: Dictionary = _preserving(runtime, profile, entries, geometry, View.WORLD_ARENA.get_center(), 100, context)
		check(repeated.ok and repeated.enemies[0].id == result.enemies[-1].id + 1, "After caller commits, another plan receives new IDs; once-only ownership remains with caller")
		if repeated.ok:
			for enemy: Dictionary in repeated.enemies: check(not seen.has(enemy.id), "Committed identities are never reused")
		# Returned state is detached from live state until the caller explicitly commits.
		var original: PackedByteArray = var_to_bytes(Encounter._snapshot(runtime))
		repeated.runtime_checkpoint.roots.clear()
		repeated.enemies[0].health = 0.0
		check(var_to_bytes(Encounter._snapshot(runtime)) == original, "Editing a later plan cannot mutate committed runtime")


func _reject(runtime: RefCounted, profile: Variant, entries: Variant, geometry: RefCounted,
		player: Variant, available: Variant, label: String, context: String = "ordinary") -> void:
	var result: Dictionary = _preserving(runtime, profile, entries, geometry, player, available, context)
	rejection_cases += 1
	check(not result.ok and not result.error.is_empty() and result.enemies.is_empty() and result.runtime_checkpoint.is_empty(), label + ": failure exposes neither a partial group nor a staged checkpoint")


func _failures(profile: Dictionary) -> void:
	var runtime: RefCounted = _runtime()
	var geometry: RefCounted = _geometry(profile.id)
	var entries: Array[Dictionary] = _entries(profile)
	var player: Vector2 = View.WORLD_ARENA.get_center()
	for index: int in [0, 1, entries.size() - 1]:
		var invalid: Array[Dictionary] = entries.duplicate(true)
		invalid[index].template_id = "missing"
		_reject(runtime, profile, invalid, geometry, player, 100, "Invalid kth template %d" % index)
		invalid = entries.duplicate(true); invalid[index].position = View.WORLD_ARENA.position - Vector2.ONE
		_reject(runtime, profile, invalid, geometry, player, 100, "Outside kth position %d" % index)
		invalid = entries.duplicate(true); invalid[index].position = player + Vector2(229.999, 0)
		_reject(runtime, profile, invalid, geometry, player, 100, "Too-close kth position %d" % index)
		invalid = entries.duplicate(true); invalid[index].position = Vector2(NAN, 250)
		_reject(runtime, profile, invalid, geometry, player, 100, "Nonfinite kth position %d" % index)
		if geometry.has_walls():
			invalid = entries.duplicate(true); invalid[index].position = geometry.snapshot().walls[0].get_center()
			_reject(runtime, profile, invalid, geometry, player, 100, "Wall kth position %d" % index)
	var bad: Array[Dictionary] = entries.duplicate(true)
	bad[1].position = bad[0].position + Vector2(1, 0)
	_reject(runtime, profile, bad, geometry, player, 100, "Valid first member then overlapping second member")
	for available: Variant in [entries.size() - 1, 0, -1, 101, true, false, float(entries.size()), 100.0, "100", null]:
		_reject(runtime, profile, entries, geometry, player, available, "Invalid/full-group-insufficient capacity " + str(available))
	for value: Variant in [null, {}, [], Vector2.INF, Vector2(NAN, 0)]:
		_reject(runtime, profile, entries, geometry, value, 100, "Invalid player")
	for value: Variant in [null, {}, [], entries.slice(0, entries.size() - 1)]:
		_reject(runtime, profile, value, geometry, player, 100, "Invalid or incomplete entry list")
	bad = entries.duplicate(true); bad.append(entries[0].duplicate(true))
	_reject(runtime, profile, bad, geometry, player, 100, "Oversized group")
	for field: String in ["template_id", "rarity", "mechanisms", "position"]:
		bad = entries.duplicate(true); bad[1].erase(field)
		_reject(runtime, profile, bad, geometry, player, 100, "Missing second-entry field " + field)
	for value: Variant in [true, 1.0, 0, -1, null]:
		bad = entries.duplicate(true); bad[1].admission_index = value
		_reject(runtime, profile, bad, geometry, player, 100, "Invalid optional admission index")
	bad = entries.duplicate(true); bad[1].rarity = "reserved"
	_reject(runtime, profile, bad, geometry, player, 100, "Factory rejects second-entry rarity")
	bad = entries.duplicate(true); bad[1].mechanisms = ["missing_mechanism"]
	_reject(runtime, profile, bad, geometry, player, 100, "Factory rejects second-entry mechanism")
	for wave: Variant in [0, -1, true, float(profile.wave), INF, NAN]:
		var invalid_profile: Dictionary = profile.duplicate(true); invalid_profile.wave = wave
		_reject(runtime, invalid_profile, entries, geometry, player, 100, "Invalid profile/wave")
	_reject(runtime, {}, entries, geometry, player, 100, "Missing profile")
	_reject(runtime, profile, entries, geometry, player, 100, "Invalid context", "death_child")
	var mismatched: RefCounted = _geometry("broken_ruins" if profile.id == "old_garden" else "old_garden")
	_reject(runtime, profile, entries, mismatched, player, 100, "Mismatched geometry")
	var corrupt: RefCounted = _runtime()
	corrupt.templates.crawler.defense_stats = {"fire_resistance": NAN}
	_reject(corrupt, profile, entries, geometry, player, 100, "Invalid template numeric stat")
	var boss: Array[Dictionary] = _boss_entry(profile)
	_reject(runtime, profile, boss, geometry, player, 0, "Boss still requires its full group", "map_boss")
	boss[0].template_id = "brute"
	_reject(runtime, profile, boss, geometry, player, 100, "Boss must match the actual profile", "map_boss")
	_reject(runtime, profile, entries, geometry, player, 100, "Boss group must contain exactly one member", "map_boss")


func _boundary_checks() -> void:
	var profile: Dictionary = Maps.compile("old_garden", [], []).profile
	var geometry: RefCounted = _geometry(profile.id)
	var runtime: RefCounted = _runtime()
	var entries: Array[Dictionary] = _entries(profile)
	entries[0].position = View.WORLD_ARENA.position + Vector2(14.0, 40.0)
	entries[1].position = entries[0].position + Vector2(24.0, 0.0)
	var player: Vector2 = entries[0].position + Vector2(0, 230.0)
	# Keep the other members far enough from this deliberately exact player limit.
	for index: int in range(2, entries.size()): entries[index].position += Vector2(1000, 0)
	var result: Dictionary = _preserving(runtime, profile, entries, geometry, player, 100)
	check(result.ok, "Exact body-boundary tangency, sum-of-radii tangency, and 230-unit player separation are admitted")
	var without_index: Array[Dictionary] = entries.duplicate(true)
	for entry: Dictionary in without_index: entry.erase("admission_index")
	check(_preserving(runtime, profile, without_index, geometry, player, 100).ok, "Optional roster admission indices may be absent")
	entries[1].position.x -= 0.001
	_reject(runtime, profile, entries, geometry, player, 100, "Actual crawler/skitter radii overlap below 24 units")
	entries = _entries(profile)
	entries[2].position.x = View.WORLD_ARENA.position.x + 21.999
	_reject(runtime, profile, entries, geometry, View.WORLD_ARENA.get_center(), 100, "22-radius ordinary footprint crosses boundary without clamping")
	var boss: Array[Dictionary] = _boss_entry(profile)
	boss[0].position.x = View.WORLD_ARENA.position.x + 27.499
	_reject(runtime, profile, boss, geometry, View.WORLD_ARENA.get_center(), 100, "27.5-radius boss footprint crosses boundary without clamping", "map_boss")
	boss[0].position.x = View.WORLD_ARENA.position.x + 27.5
	check(_preserving(runtime, profile, boss, geometry, View.WORLD_ARENA.get_center(), 1, "map_boss").ok, "Boss exact-radius boundary tangency is admitted")


func _owner_snapshot(owner: Variant) -> PackedByteArray:
	if not owner is PreparedEntry: return var_to_bytes(owner)
	return var_to_bytes([owner.phase(), owner._context, owner.landmarks(), owner.bounds(),
		owner.geometry_ref().get_instance_id() if owner.geometry_ref() != null else 0,
		owner._release.is_valid()])


func _native_snapshot(geometry: RefCounted) -> PackedByteArray:
	var body_ids: Array[int] = []
	var shape_ids: Array[int] = []
	for body: RID in geometry._bodies: body_ids.append(body.get_id())
	for shape: RID in geometry._shapes: shape_ids.append(shape.get_id())
	# The reusable circle's radius is query scratch, not collision authority.
	return var_to_bytes([geometry.get_instance_id(), geometry.snapshot(), geometry._routes,
		geometry._space.get_id(), body_ids, shape_ids, geometry._polygons,
		geometry._native_probes, geometry._probe_contours, geometry._study_active,
		geometry._native_ready, geometry._query_circle.get_instance_id() if geometry._query_circle != null else 0])


func _native_preserving(runtime: RefCounted, profile: Variant, entries: Variant,
		geometry: RefCounted, player: Variant, available: Variant, context: String,
		owner: Variant) -> Dictionary:
	var owner_before := _owner_snapshot(owner)
	var geometry_before := _native_snapshot(geometry)
	var result := _preserving(runtime, profile, entries, geometry, player, available, context, owner)
	check(_owner_snapshot(owner) == owner_before, "Native camp plan preserves the prepared owner, phase, context, bounds and detached routes")
	check(_native_snapshot(geometry) == geometry_before, "Native camp plan preserves exact geometry identity, contours, readiness, native resource handles and cached routes")
	return result


func _native_reject(runtime: RefCounted, profile: Variant, entries: Variant,
		geometry: RefCounted, player: Variant, available: Variant, label: String,
		context: String, owner: Variant) -> void:
	var result := _native_preserving(runtime, profile, entries, geometry, player, available, context, owner)
	rejection_cases += 1
	native_rejections.append(label + " / " + context)
	check(not result.ok and not result.error.is_empty() and result.enemies.is_empty()
		and result.runtime_checkpoint.is_empty(), label + ": native rejection exposes no partial actor group or checkpoint")


func _released(owner: RefCounted, geometry: RefCounted, label: String) -> void:
	check(owner.phase() == "cancelled" and owner.geometry_ref() == null and owner.landmarks().is_empty()
		and owner._context.is_empty() and not owner._release.is_valid(), label + ": prepared owner is cancelled and detached")
	check(not geometry._space.is_valid() and geometry._bodies.is_empty() and geometry._shapes.is_empty()
		and geometry._query_circle == null and geometry._body_contours.is_empty()
		and geometry._native_probes.is_empty() and geometry._polygons.is_empty()
		and not geometry.physics_ready(), label + ": all native resource ownership is released")


func _native_group(profile: Dictionary, runtime: RefCounted, entries: Array[Dictionary],
		geometry: RefCounted, landmarks: Dictionary, owner: RefCounted, context: String) -> void:
	var before := Encounter._snapshot(runtime)
	var result := _native_preserving(runtime, profile, entries, geometry, landmarks.entry, entries.size(), context, owner)
	check(result.ok and result.error.is_empty() and result.enemies.size() == entries.size(),
		"Native whole group stages at exact capacity: " + profile.summary + " / " + context)
	if not result.ok: return
	positive_groups += 1
	native_groups += 1
	var expected = Runtime.new(runtime.templates)
	Encounter._restore(expected, before.duplicate(true))
	var seen: Dictionary = {}
	for index: int in range(entries.size()):
		var entry: Dictionary = entries[index]
		var enemy: Dictionary = result.enemies[index]
		var reference: Dictionary = Admission.create_root(expected, profile, entry.template_id,
			profile.wave, entry.position, context, entry.rarity, entry.mechanisms, true)
		check(reference.ok and var_to_bytes(reference.enemy) == var_to_bytes(enemy), "Every native actor field equals the actual map admission factory")
		check(enemy.id == before.next_id + index + 1 and not seen.has(enemy.id), "Native staged identities remain sequential and unique")
		seen[enemy.id] = true
		check(enemy.pos == entry.position and enemy.spawn == 0.6 and enemy.wave == profile.wave,
			"Native actor keeps the real roster position, tier wave and spawn delay")
		var radius: float = float(Monsters.SPECIES[enemy.kind].radius) * (1.25 if context == "map_boss" else 1.0)
		check(enemy.radius == radius and geometry.is_clear(enemy.pos, radius)
			and enemy.pos.distance_to(landmarks.entry) >= Camps.PLAYER_CLEARANCE, "Native collision authority admits the actual species radius and player clearance")
		check(enemy.root_id == enemy.id and enemy.generation == 0 and enemy.reward_eligible, "Native staged actors retain reward-eligible root lineage")
		if not profile.normal_ids.is_empty(): check(enemy.has("encounter_source"), "Native normal modifiers retain actual source metadata")
		if profile.special_ids.has("elemental_aegis"): check(enemy.has("map_defense_source"), "Native aegis retains actual source metadata")
		if context == "map_boss":
			check(enemy.template_id == "rift_warden" and enemy.map_boss_attack_id == "ruins_garden_slam"
				and Bosses.for_enemy(enemy).map_id == "ruins_garden", "Native boss retains its strict new-map attack identity in the real resolver")
		else:
			for special: String in profile.special_ids:
				if MapCatalog.SPECIAL[special].get("template", "") == enemy.template_id:
					special_templates[special] = int(special_templates.get(special, 0)) + 1
	check(var_to_bytes(result.runtime_checkpoint) == var_to_bytes(Encounter._snapshot(expected)), "Native returned checkpoint equals the real factory authority byte for byte")
	check(result.runtime_checkpoint.queue == before.queue and result.runtime_checkpoint.trace == before.trace, "Native planning preserves pending descendants and prior trace")
	for id: int in before.roots:
		check(result.runtime_checkpoint.roots[id] == before.roots[id], "Native planning preserves every existing processed root and lineage")
	Encounter._restore(runtime, result.runtime_checkpoint.duplicate(true))
	var repeated := _native_preserving(runtime, profile, entries, geometry, landmarks.entry, 100, context, owner)
	check(repeated.ok and repeated.enemies[0].id == result.enemies[-1].id + 1, "Native caller commit advances IDs without consuming the entry owner")
	if not repeated.ok: return
	for enemy: Dictionary in repeated.enemies: check(not seen.has(enemy.id), "Native committed IDs are never reused")
	var committed := var_to_bytes(Encounter._snapshot(runtime))
	repeated.runtime_checkpoint.roots.clear()
	repeated.enemies[0].health = 0.0
	check(var_to_bytes(Encounter._snapshot(runtime)) == committed, "Returned native enemies and later checkpoint cannot mutate committed runtime")


func _native_positive(profile: Dictionary, geometry: RefCounted, landmarks: Dictionary, owner: RefCounted) -> void:
	var state = CampState.new()
	var profile_before := var_to_bytes(profile)
	var landmarks_before := var_to_bytes(landmarks)
	seed(116430)
	var expected_random: Array[int] = [randi(), randi(), randi()]
	seed(116430)
	var selected: Dictionary = state.begin(profile, landmarks, 116430, Monsters.CURRENT_ROLL_POLICY)
	check([randi(), randi(), randi()] == expected_random, "Actual native camp roster selection leaves global RNG untouched")
	check(selected.ok and var_to_bytes(profile) == profile_before and var_to_bytes(landmarks) == landmarks_before,
		"Actual native camp roster selection preserves the compiled profile and prepared landmarks")
	if not selected.ok: return
	check(profile.id == "ruins_garden" and profile.boss_attack_id == "ruins_garden_slam"
		and Bosses.profile_reason(profile).is_empty(), "Each native tier/modifier profile has its explicit boss binding")
	var roster_before := var_to_bytes(state.checkpoint())
	var runtime: RefCounted = _runtime()
	var groups_before := native_groups
	for camp: Dictionary in landmarks.camps:
		var entries: Array[Dictionary] = state.entries(camp.id)
		check(entries.size() == 8, "Actual native source camp supplies all eight roots")
		_native_group(profile, runtime, entries, geometry, landmarks, owner, "ordinary")
	var boss: Array[Dictionary] = [{"template_id": profile.boss_id, "rarity": "", "mechanisms": [], "position": landmarks.boss.center}]
	_native_group(profile, runtime, boss, geometry, landmarks, owner, "map_boss")
	check(native_groups == groups_before + 4, "Every allowed native profile admits all three ordinary groups and its boss")
	check(var_to_bytes(state.checkpoint()) == roster_before and var_to_bytes(landmarks) == landmarks_before,
		"Native camp planning leaves the actual camp roster and all prepared routes byte-identical")
	native_profiles.append({"tier": profile.journey_tier, "wave": profile.wave,
		"normal_ids": profile.normal_ids, "special_ids": profile.special_ids,
		"staged_groups": native_groups - groups_before})


func _native_invalid_groups(profile: Dictionary, geometry: RefCounted, landmarks: Dictionary, owner: RefCounted) -> void:
	var runtime: RefCounted = _runtime()
	var state = CampState.new()
	check(state.begin(profile, landmarks, 116430, Monsters.CURRENT_ROLL_POLICY).ok, "Native rejection fixture uses the actual complete roster")
	var entries: Array[Dictionary] = state.entries(landmarks.camps[0].id)
	for index: int in [0, 1, entries.size() - 1]:
		var invalid: Array[Dictionary] = entries.duplicate(true)
		invalid[index].template_id = "missing"
		_native_reject(runtime, profile, invalid, geometry, landmarks.entry, 100, "Native invalid kth template %d" % index, "ordinary", owner)
		for point: Vector2 in [Layout.WORLD_BOUNDS.position - Vector2.ONE, landmarks.entry + Vector2(229.999, 0), Vector2(NAN, 250), geometry._native_probes[0]]:
			invalid = entries.duplicate(true); invalid[index].position = point
			_native_reject(runtime, profile, invalid, geometry, landmarks.entry, 100, "Native invalid kth body position %d: %s" % [index, point], "ordinary", owner)
	var overlapping: Array[Dictionary] = entries.duplicate(true)
	overlapping[1].position = overlapping[0].position + Vector2.ONE
	_native_reject(runtime, profile, overlapping, geometry, landmarks.entry, 100, "Native late overlapping member", "ordinary", owner)
	for available: Variant in [7, 0, -1, 101, true, 8.0, "100", null]:
		_native_reject(runtime, profile, entries, geometry, landmarks.entry, available, "Native invalid or insufficient capacity " + str(available), "ordinary", owner)
	var mixed: Dictionary = profile.duplicate(true); mixed.boss_attack_id = "garden_slam"
	var boss: Array[Dictionary] = [{"template_id": profile.boss_id, "rarity": "", "mechanisms": [], "position": landmarks.boss.center}]
	_native_reject(runtime, mixed, boss, geometry, landmarks.entry, 1, "Native profile cannot borrow old garden boss binding", "map_boss", owner)
	_native_reject(runtime, profile, boss, geometry, landmarks.entry, 0, "Native boss requires one whole slot", "map_boss", owner)
	boss[0].template_id = "brute"
	_native_reject(runtime, profile, boss, geometry, landmarks.entry, 1, "Native boss requires the profile template", "map_boss", owner)
	_native_reject(runtime, profile, entries, geometry, landmarks.entry, 100, "Native boss context cannot admit an ordinary group", "map_boss", owner)


func _native_checks(selections: Array) -> void:
	check(MapCatalog.MAPS.size() == LEGACY_MAP_IDS.size() + 1 and MapCatalog.MAPS.has("ruins_garden")
		and MapCatalog.MAPS.ruins_garden.native_entry, "All five catalogue maps have an explicit exercised branch; the new map requires native entry")
	var bounds := Layout.WORLD_BOUNDS
	var source: Dictionary = Layout.layout("ruins_garden", bounds)
	var assembly: Dictionary = NativeSession.layout(bounds, "ruins_garden")
	check(source.ok and assembly.ok, "Native camp fixture reads the actual new-map layout and original modular assembly")
	if not source.ok or not assembly.ok: return
	var source_before := var_to_bytes(source)
	var assembly_before := var_to_bytes(assembly)
	var profile: Dictionary = Maps.compile_normal("ruins_garden", 1, [], []).profile
	var runtime: RefCounted = _runtime()
	var state = CampState.new()
	check(state.begin(profile, source.landmarks, 116430, Monsters.CURRENT_ROLL_POLICY).ok, "Native ownership rejection uses an actual lawful camp roster")
	var entries: Array[Dictionary] = state.entries(source.landmarks.camps[0].id)
	var boss: Array[Dictionary] = [{"template_id": profile.boss_id, "rarity": "", "mechanisms": [], "position": source.landmarks.boss.center}]
	var fallback = Geometry.new()
	check(fallback.configure("old_garden", View.WORLD_ARENA), "Rectangle rejection fixture begins with valid historical geometry")
	var fallback_before := var_to_bytes([fallback.snapshot(), fallback._routes])
	check(not fallback.configure("ruins_garden", bounds) and not fallback.configure_exploration("ruins_garden", bounds)
		and var_to_bytes([fallback.snapshot(), fallback._routes]) == fallback_before, "Both rectangle APIs reject native identity without geometry mutation")
	for context: String in ["ordinary", "map_boss"]:
		var group: Array[Dictionary] = entries if context == "ordinary" else boss
		_reject(runtime, profile, group, fallback, source.landmarks.entry, 100, "Native map cannot use historical rectangle geometry", context)
	var pending = NativeGeometry.new()
	var pending_owner = PreparedEntry.new(NATIVE_CONTEXT.to_utf8_buffer())
	check(pending.install(bounds, assembly.polygons, source.landmarks.entry, assembly.presentation, "ruins_garden").ok,
		"Unready fixture installs the unchanged real native assembly")
	check(pending_owner.attach(pending, Callable(pending, "_release_native")), "Unready fixture owns its detached native candidate")
	check(not pending.physics_ready(), "Installed native candidate is genuinely unready before physics synchronization")
	for context: String in ["ordinary", "map_boss"]:
		_native_reject(runtime, profile, entries if context == "ordinary" else boss, pending,
			source.landmarks.entry, 100, "Preparing native owner", context, pending_owner)
	check(pending_owner.ready(source.landmarks, bounds) and pending_owner.begin_use(NATIVE_CONTEXT.to_utf8_buffer()).is_empty(),
		"Unready geometry is tested even with an active typed owner")
	for context: String in ["ordinary", "map_boss"]:
		_native_reject(runtime, profile, entries if context == "ordinary" else boss, pending,
			source.landmarks.entry, 100, "Active owner with unsynchronized native geometry", context, pending_owner)
	pending_owner.finish_use(false)
	_released(pending_owner, pending, "Rejected unready native candidate")
	var geometry = NativeGeometry.new()
	var other_geometry = NativeGeometry.new()
	var owner = PreparedEntry.new(NATIVE_CONTEXT.to_utf8_buffer())
	var other_owner = PreparedEntry.new(NATIVE_CONTEXT.to_utf8_buffer())
	for pair: Array in [[geometry, owner], [other_geometry, other_owner]]:
		check(pair[0].install(bounds, assembly.polygons, source.landmarks.entry, assembly.presentation, "ruins_garden").ok,
			"Install actual native camp collision authority")
		check(pair[1].attach(pair[0], Callable(pair[0], "_release_native")), "Prepared owner retains its exact native candidate")
	await physics_frame
	await physics_frame
	var native_ready: bool = geometry.physics_ready() and other_geometry.physics_ready()
	check(native_ready, "Both real native spaces synchronize every convex collision piece")
	if not native_ready:
		owner.cancel(); other_owner.cancel()
		_released(owner, geometry, "Failed native synchronization")
		_released(other_owner, other_geometry, "Failed alternate native synchronization")
		return
	var routes: Dictionary = NativeRoutes.prepare(source.landmarks, geometry)
	check(routes.ok and routes.landmarks.route_segments.size() == 14, "Actual native route adapter supplies the fourteen complete detour segments")
	if not routes.ok:
		owner.cancel(); other_owner.cancel()
		_released(owner, geometry, "Failed native route preparation")
		_released(other_owner, other_geometry, "Failed alternate route preparation")
		return
	var vertex_count := 0
	for polygon: PackedVector2Array in geometry.snapshot().module_polygons: vertex_count += polygon.size()
	check(geometry.snapshot().id == "ruins_garden" and geometry.snapshot().source_map_id == "ruins_garden"
		and geometry.snapshot().module_polygons.size() == 4 and vertex_count == 99, "Native camp authority retains its own identity and all four original 99-vertex contours")
	for index: int in range(routes.landmarks.route_segments.size()):
		var route: Dictionary = routes.landmarks.route_segments[index]
		var radius: float = 51.0 if index in [0, 1, 11, 12] else 36.0
		check(route.width == 72.0 and geometry.is_clear(route.from, radius) and geometry.is_clear(route.to, radius)
			and not geometry.sweep(route.from, route.to, radius).hit and not geometry.sweep(route.to, route.from, radius).hit,
			"Actual native camp routes clear both directions at their original authored width")
	check(other_owner.ready(routes.landmarks, bounds) and other_owner.begin_use(NATIVE_CONTEXT.to_utf8_buffer()).is_empty(),
		"Wrong-owner fixture is independently ready and active for a different real same-map geometry")
	var other_before := _native_snapshot(other_geometry)
	for context: String in ["ordinary", "map_boss"]:
		var group: Array[Dictionary] = entries if context == "ordinary" else boss
		_native_reject(runtime, profile, group, geometry, source.landmarks.entry, 100, "Ready geometry without prepared owner", context, null)
		_native_reject(runtime, profile, group, geometry, source.landmarks.entry, 100, "Forged native ready dictionary", context,
			{"validated": true, "ready": true, "phase": "in_use", "geometry": geometry, "landmarks": routes.landmarks})
		_native_reject(runtime, profile, group, geometry, source.landmarks.entry, 100, "Wrong native owner despite matching map and ready state", context, other_owner)
		_native_reject(runtime, profile, group, geometry, source.landmarks.entry, 100, "Synchronized geometry with preparing owner", context, owner)
	check(_native_snapshot(other_geometry) == other_before, "Wrong-owner rejections leave the other native collision authority untouched")
	check(owner.ready(routes.landmarks, bounds), "Correct typed owner retains the real prepared routes")
	for context: String in ["ordinary", "map_boss"]:
		_native_reject(runtime, profile, entries if context == "ordinary" else boss, geometry, source.landmarks.entry,
			100, "Ready owner cannot bypass begin_use", context, owner)
	check(owner.begin_use(NATIVE_CONTEXT.to_utf8_buffer()).is_empty(), "Correct prepared native owner enters active use")
	var special_selections: Array = [[]]
	for id: String in MapCatalog.SPECIAL: special_selections.append([id])
	var tier_profiles: Dictionary = {}
	var rejected_tier_modifiers := 0
	for tier: int in range(1, 4):
		tier_profiles[tier] = 0
		for normal_ids: Array in selections:
			for special_ids: Array in special_selections:
				var compiled: Dictionary = Maps.compile_normal("ruins_garden", tier, normal_ids, special_ids)
				var allowed: bool = special_ids.is_empty() or [1, 4, 8][tier - 1] >= int(MapCatalog.SPECIAL[special_ids[0]].minimum_wave)
				check(compiled.ok == allowed, "Native tier compiler honors every actual modifier wave gate")
				if not compiled.ok:
					rejected_tier_modifiers += 1
					continue
				tier_profiles[tier] += 1
				_native_positive(compiled.profile, geometry, routes.landmarks, owner)
	check(tier_profiles == {1: 10, 2: 30, 3: 50}, "Native coverage includes all three tiers and every lawful selected normal/special combination")
	for id: String in ["frost_patrol", "storm_patrol", "chaos_patrol"]:
		check(int(special_templates.get(id, 0)) > 0, "Actual native rosters exercise special-template admission: " + id)
	_native_invalid_groups(profile, geometry, routes.landmarks, owner)
	check(var_to_bytes(source) == source_before and var_to_bytes(assembly) == assembly_before,
		"Every native success and failure preserves the original authored layout and assembly")
	report["native_contract"] = {"tiers": tier_profiles, "disallowed_tier_modifier_profiles": rejected_tier_modifiers,
		"route_segments": routes.landmarks.route_segments.size(), "contours": 4, "vertices": vertex_count}
	owner.finish_use(false)
	other_owner.finish_use(false)
	_released(owner, geometry, "Completed camp fixture native candidate")
	_released(other_owner, other_geometry, "Completed alternate owner candidate")
	for context: String in ["ordinary", "map_boss"]:
		_native_reject(runtime, profile, entries if context == "ordinary" else boss, geometry, source.landmarks.entry,
			100, "Released native owner cannot be replayed", context, owner)
	owner.finish_use(false); other_owner.finish_use(false)
	_released(owner, geometry, "Repeated native cleanup is idempotent")
	_released(other_owner, other_geometry, "Repeated alternate cleanup is idempotent")
